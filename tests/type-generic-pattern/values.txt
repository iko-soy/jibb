  [%text value] value
  [%node event kids] (event "click")
++ nested ?- x
  [%text value] value
  [%node event kids] ?- kids
    ~ "empty"
    [item remainder] ?- item
      [%text value] value
      [%node action rest] (action "click")
++ fallback ?- x
  [%text value] value
  * "node"
++ whole-bind ?- x
  other other
++ whole-tail ?- x
  [%node payload] "node"
  * "other"
