=/  m  (strand ,vase)
;<  =bowl:spider  bind:m  get-bowl
=/  now=@da  now.bowl
=/  pre=path  (en-beam [our.bowl %jibb-fixtures da+now] /)
=/  actual=vase  .^(vase %ca (weld pre /sur/tables/hoon))
=/  reader=vase  .^(vase %ca (weld pre /lib/jibb-import/hoon))
=/  jibb=vase  .^(vase %ca (weld pre /lib/jibb/hoon))
=/  paths=(list path)
  ~[/sur/tables/hoon /sur/ford-types/hoon /lib/ford-helpers/hoon /sur/ford-base/hoon]
=/  sources=(list [at=path source=@t])
  (turn paths |=(at=path [at .^(@t %cx (weld pre at))]))
=/  read-bundle=vase  (slap reader (ream 'read-module-bundle'))
=/  entries=(each (list [module=term name=term result=*]) @t)
  !<((each (list [module=term name=term result=*]) @t) (slym read-bundle [%tables /sur/tables/hoon sources]))
?:  ?=(%| -.entries)  (pure:m !>(p.entries))
=/  sample=(list [module=term name=term result=*])
  (skim p.entries |=(entry=[module=term name=term result=*] =(%sample name.entry)))
?>  ?=([* ~] sample)
=/  read-name=vase  (slap reader (ream 'read-named'))
=/  clay-sample=vase  (slym read-name [%tables %sample -.actual])
?.  =(+.clay-sample result.i.sample)
  (pure:m !>('Clay sample differs from source reconstruction'))
=/  check=vase  (slap jibb (ream 'infer-jibb-file-detailed'))
=/  at=path  /ui/ford-imports/jibb
=/  source=@t  .^(@t %cx (weld pre /ui/ford-imports/jibb))
=/  checked=(each * [message=@t overlap=*])
  !<((each * [message=@t overlap=*]) (slym check [at source p.entries ~ ~ ~[at] ~ ~ ~ ~]))
?:  ?=(%| -.checked)  (pure:m !>(message.p.checked))
(pure:m !>('Ford generic full check and Clay sample agree'))
